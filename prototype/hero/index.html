weather.ts
import { openai } from '@ai-sdk/openai';
import { Agent } from '@balsa/core/agent';   // one import per subsystem
import { createTool } from '@balsa/core/tools';
import { z } from 'zod';

const weather = createTool({
  description: 'Looks up the current weather for a city.',
  inputSchema: z.object({ city: z.string() }),
  execute: ({ city }) => ({ city, celsius: 18 }),
}); // a tool is four fields

const agent = new Agent({
  name: 'assistant',
  instructions: 'You are concise.',
  model: openai.chat('gpt-4o-mini'),
  tools: { weather },
});

for await (const chunk of agent.stream('Weather in Paris?')) {
  if (chunk.type === 'text-delta') process.stdout.write(chunk.textDelta);
}
